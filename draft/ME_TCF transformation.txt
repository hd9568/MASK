5. split y_s
### on sparse axis split, we spawn a child format
format = (
	x_o, x, d, f, M
	y_s, y, s, v, n
	x_i, x, d, f, TM
	child = (
		y_so, y, d, f, N (N * TN = n)
		y_si, y, d, f, TN
		x_i, x, d, f, TM
	)

array:

val = x_o_y_so[nnz_b]y_si[TN]x_i[TM]
# y_s_idx = x_o_y_s[nnz]
y_s_idx = x_o_y_so[nnz_b]y_si[TN]
x_o_off = x_o[M]
b = y[n]z[k]

kernel:
# for i in 0..M:
for i in dense_iter(x_o):
	# l, r = x_o_off[i], x_o_off[i+1]
	# for j in l..r:
	for j in sparse_iter(x_o_off, i, y_so):
		atomic_dense(val[j], b[y_s_idx[j]])

6. coo-ize
### on coo-ization, we spawn a child format
format = (
	x_o, x, d, f, M
	y_s, y, s, v, n
	x_i, x, d, f, TM
	child = (
		y_so, y, d, f, N (N * TN = n)
		y_si, y, d, f, TN
		x_i, x, d, f, TM
		child = atomic_coo(
			x_i, x, d, f, TM
			y_si, y, d, f, TN
		)
	)
)

array:
# val = x_o_y_so[nnz_b]y_si[TN]x_i[TM]
(new) coo_val = x_o_y_so_coo[nnz]
(new) coo_idx = x_o_y_so_coo[nnz]
(new) coo_off = x_o_y_so[nnz_b]
y_s_idx = x_o_y_so[nnz_b]y_si[TN]
x_o_off = x_o[M]
b = y[n]z[k]

kernel:
for i in dense_iter(x_o):
	# l, r = x_o_off[i], x_o_off[i+1]
	# for j in l..r:
	for j in sparse_iter(x_o_off, i, y_so):
		# coo_l, coo_r = coo_off[j], coo_off[j+1]
		# atomic_coo(coo_val[coo_l:coo_r], coo_idx[coo_l:coo_r], b[y_s_idx[j]])
		atomic_coo(coo_off, coo_val, coo_idx, j, b[y_s_idx[j]])
